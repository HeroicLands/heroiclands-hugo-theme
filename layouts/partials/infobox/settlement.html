{{- /* Settlement infobox. A floated card rendered at the top of a settlement page's main content column (type: place, subType: settlement). Mirrors the polity infobox: a shortcode-driven profile card whose location row resolves to real pages and renders as links when a match exists. Rows: - Name — name.full (fallback: title) - Population — data.population, rendered verbatim ("~450,000") - Region — the political chain. Climbs the settlement's own `data.parents`, preferring an entry that resolves to a polity, then that polity's own `data.parents` for its parent polity (nome/province → empire), and renders the chain root→leaf as links joined by " — ", e.g. "The Empire of Ta'Kheperu — The Capital Nome". Every section is guarded so missing data silently disappears. */ -}} {{- $ctx := . -}} {{- $data := dict -}}{{- with .Params.data }}{{ $data = . }}{{ end -}} {{- $name := "" -}}{{- with .Params.name.full }}{{ $name = . }}{{ end -}} {{- if not $name }}{{ $name = .Title }}{{ end -}} {{- $pop := $data.population -}}{{/* integer; 0 is a valid value (e.g. a necropolis), so guard with `ne nil` rather than `with` */}} {{- /* First parent of a settlement's own `data.parents` that resolves to a polity. */ -}} {{- $findPolity := "" -}} {{- range $data.parents -}} {{- if not $findPolity -}} {{- $pg := first 1 (where (where $ctx.Site.RegularPages "Params.shortcode" .) "Params.subType" "polity") -}} {{- if $pg }}{{ $findPolity = . }}{{ end -}} {{- end -}} {{- end -}} {{- /* Build the political chain by climbing `data.parents`. Start at the settlement's immediate polity and walk up each polity's own first parent polity. Prepend each page so the collected order is root→leaf. Fixed-depth loop with a guard (real chains are 1–2 deep; 6 is ample) since Hugo has no while loop. */ -}} {{- $chain := slice -}} {{- $cur := $findPolity -}} {{- range (seq 6) -}} {{- with $cur -}} {{- $pg := first 1 (where $ctx.Site.RegularPages "Params.shortcode" .) -}} {{- with index $pg 0 -}} {{- $chain = (slice .) | append $chain -}} {{- $cur = "" -}} {{- range .Params.data.parents -}} {{- if not $cur -}} {{- $ppg := first 1 (where (where $ctx.Site.RegularPages "Params.shortcode" .) "Params.subType" "polity") -}} {{- if $ppg }}{{ $cur = . }}{{ end -}} {{- end -}} {{- end -}} {{- else -}} {{- $cur = "" -}} {{- end -}} {{- end -}} {{- end -}}

{{- end }} {{- /* Location row. Prefer the political chain (climbed above). When a settlement has no polity ancestry — e.g. K'ich'chik cities, whose city-states are modeled as regions rather than polities — fall back to resolving the region entries of the settlement's own `data.parents` so the row is still populated. */ -}} {{- $locParts := slice -}} {{- if gt (len $chain) 0 -}} {{- range $chain -}} {{- $locParts = $locParts | append (printf `%s` .Permalink .LinkTitle) -}} {{- end -}} {{- else -}} {{- range $code := $data.parents -}} {{- $pg := first 1 (where (where $ctx.Site.RegularPages "Params.shortcode" $code) "Params.subType" "region") -}} {{- with index $pg 0 -}} {{- $locParts = $locParts | append (printf `%s` .Permalink .LinkTitle) -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt (len $locParts) 0 }}
